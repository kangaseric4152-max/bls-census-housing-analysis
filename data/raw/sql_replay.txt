select
  cbsa as cbsa_code,
  cast(regexp_extract(year_col, 'POPESTIMATE(\d{4})', 1) as int) as year,
  population
from (
  select *
  from census_2020_raw
  where lsad = 'Metropolitan Statistical Area'
